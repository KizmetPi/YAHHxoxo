    <script>
        let highestZ = 10;

        document.querySelectorAll('.MTAB').forEach(tab => {
            const bar = tab.querySelector('.Tbar');
            const resizer = tab.querySelector('.resize-handle');

            // Bring active window to front
            const bringToFront = () => {
                highestZ++;
                tab.style.zIndex = highestZ;
            };
            tab.addEventListener('mousedown', bringToFront);
            tab.addEventListener('touchstart', bringToFront, { passive: true });

            // --- DRAGGING LOGIC (Mouse & Touch) ---
            let isDragging = false;
            let startX, startY, initialLeft, initialTop;

            const startDrag = (e) => {
                isDragging = true;
                bringToFront();
                const point = e.touches ? e.touches[0] : e;
                startX = point.clientX;
                startY = point.clientY;
                initialLeft = tab.offsetLeft;
                initialTop = tab.offsetTop;
                tab.style.setProperty('left', initialLeft + 'px', 'important');
                tab.style.setProperty('top', initialTop + 'px', 'important');
            };

            const onDrag = (e) => {
                if (!isDragging) return;
                const point = e.touches ? e.touches[0] : e;
                const dx = point.clientX - startX;
                const dy = point.clientY - startY;
                tab.style.setProperty('left', (initialLeft + dx) + 'px', 'important');
                tab.style.setProperty('top', (initialTop + dy) + 'px', 'important');
            };

            const stopDrag = () => {
                isDragging = false;
            };

            bar.addEventListener('mousedown', startDrag);
            bar.addEventListener('touchstart', startDrag, { passive: true });
            document.addEventListener('mousemove', onDrag);
            document.addEventListener('touchmove', onDrag, { passive: true });
            document.addEventListener('mouseup', stopDrag);
            document.addEventListener('touchend', stopDrag);

            // --- RESIZING LOGIC (Mouse & Touch) ---
            let isResizing = false;
            let resizeStartX, resizeStartY, startWidth, startHeight;

            const startResize = (e) => {
                e.stopPropagation();
                isResizing = true;
                bringToFront();
                const point = e.touches ? e.touches[0] : e;
                resizeStartX = point.clientX;
                resizeStartY = point.clientY;
                startWidth = tab.offsetWidth;
                startHeight = tab.offsetHeight;
            };

            const onResize = (e) => {
                if (!isResizing) return;
                const point = e.touches ? e.touches[0] : e;
                const newWidth = Math.max(240, startWidth + (point.clientX - resizeStartX));
                const newHeight = Math.max(180, startHeight + (point.clientY - resizeStartY));
                tab.style.width = newWidth + 'px';
                tab.style.height = newHeight + 'px';
            };

            const stopResize = () => {
                isResizing = false;
            };

            resizer.addEventListener('mousedown', startResize);
            resizer.addEventListener('touchstart', startResize, { passive: true });
            document.addEventListener('mousemove', onResize);
            document.addEventListener('touchmove', onResize, { passive: true });
            document.addEventListener('mouseup', stopResize);
            document.addEventListener('touchend', stopResize);
        });
    </script>
